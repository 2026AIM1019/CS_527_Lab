% Test 5: FIR filter response
% 3-tap FIR: y[n] = x[n] + 2*x[n-1] + x[n-2]
% N is stored at Data[0]
% Input starts at Data[4]
% Output starts at Data[16]
% Missing previous samples are treated as 0

x1 = 0      % index
x2 = [0]    % N
x3 = 4      % input address
x4 = 16     % output address
x8 = 0      % previous sample x[n-1]
x9 = 0      % previous-previous sample x[n-2]

.loop
x5 = x1 - x2
BEQ .exit

x6 = [x3]   % current sample
x7 = x6 * 1
x10 = x8 * 2
x11 = x9 * 1
x12 = x7 + x10
x12 = x12 + x11
[x4] = x12

x9 = x8 + 0
x8 = x6 + 0
x3 = x3 + 4
x4 = x4 + 4
x1 = x1 + 1
BAL .loop

.exit
